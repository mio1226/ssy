    private final ViolationRecordMapper violationRecordMapper;

    private final SysUserMapper sysUserMapper;
    private final HardDiskMapper hardDiskMapper;

    public String getRuleValue(String ruleKey) {
        RuleConfig config = ruleConfigMapper.selectOne(
                new LambdaQueryWrapper<RuleConfig>()
                        .eq(RuleConfig::getRuleKey, ruleKey)
                        .eq(RuleConfig::getStatus, 1));
        return config != null ? config.getRuleValue() : null;
    }

    public boolean getRuleBoolean(String ruleKey) {
        return "1".equals(getRuleValue(ruleKey));
    }

    @Transactional
    public void recordViolation(Long userId, String username, Long diskId, Long recordId, String type, String description) {
        ViolationRecord violation = new ViolationRecord();
        violation.setUserId(userId);
        violation.setUsername(username);
        violation.setDiskId(diskId);
        violation.setRecordId(recordId);
        violation.setType(type);
        violation.setDescription(description);
        violation.setStatus(0);
        violationRecordMapper.insert(violation);
        log.info("违规记录已生成: type={}, userId={}, diskId={}, desc={}", type, userId, diskId, description);
    }

    /**
     * 创建或更新违规记录（已存在未处理的则更新，不存在则新增）
     */
    @Transactional
    public void upsertViolation(Long userId, String username, Long diskId, Long recordId, String type, String description) {
        ViolationRecord existing = violationRecordMapper.selectOne(
                new LambdaQueryWrapper<ViolationRecord>()
                        .eq(ViolationRecord::getDiskId, diskId)
                        .eq(ViolationRecord::getType, type)
                        .eq(ViolationRecord::getStatus, 0));
        if (existing != null) {
            existing.setDescription(description);
            if (recordId != null) existing.setRecordId(recordId);
            if (userId != null) existing.setUserId(userId);
            if (username != null) existing.setUsername(username);
            violationRecordMapper.updateById(existing);
            log.info("违规记录已更新: type={}, diskId={}, desc={}", type, diskId, description);
        } else {
            recordViolation(userId, username, diskId, recordId, type, description);
        }
    }

    /**
     * 定时检查超时记录
     */
    @Scheduled(fixedDelay = 60000)
    @Transactional
    public void checkTimeout() {
        RuleConfig config = ruleConfigMapper.selectOne(
                new LambdaQueryWrapper<RuleConfig>()
                        .eq(RuleConfig::getRuleKey, "timeout_days"));
        if (config == null || config.getRuleValue() == null || config.getRuleValue().trim().isEmpty()) return;
        if (config.getStatus() == null || config.getStatus() != 1) return;

        int timeoutDays;
        try {
            timeoutDays = Integer.parseInt(config.getRuleValue());
        } catch (NumberFormatException e) {
            log.warn("timeout_days 配置值不是合法整数: {}", config.getRuleValue());
            return;
        }

        LocalDateTime deadline = LocalDateTime.now().minusDays(timeoutDays);

        List<DiskUsageRecord> timeoutRecords = usageRecordMapper.selectList(
                new LambdaQueryWrapper<DiskUsageRecord>()
                        .eq(DiskUsageRecord::getStatus, 1)
                        .lt(DiskUsageRecord::getOutTime, deadline));

        for (DiskUsageRecord record : timeoutRecords) {
            String username = "";
            SysUser user = sysUserMapper.selectById(record.getOperatorId());
            if (user != null) username = user.getUsername();

            upsertViolation(record.getOperatorId(), username, record.getDiskId(), record.getId(),
                    "timeout", "硬盘出库超时，出库时间：" + record.getOutTime());
            log.info("硬盘ID: {} 出库超时，已生成违规记录", record.getDiskId());
        }
    }

    /**
     * 定时检查不完整数据记录
     */
    @Scheduled(fixedDelay = 60000)
    @Transactional
    public void checkIncompleteData() {
        if (!getRuleBoolean("incomplete_data_check")) return;

        List<HardDisk> incompleteDisks = hardDiskMapper.selectList(
                new LambdaQueryWrapper<HardDisk>()
                        .and(w -> w
                                .isNull(HardDisk::getModel).or().eq(HardDisk::getModel, "")
                                .or().isNull(HardDisk::getSn).or().eq(HardDisk::getSn, "")
                                .or().isNull(HardDisk::getCapacity))
        );
        for (HardDisk disk : incompleteDisks) {
            StringBuilder desc = new StringBuilder("硬盘信息不完整");
            if (disk.getModel() == null || disk.getModel().isEmpty()) desc.append("，缺少型号");
            if (disk.getSn() == null || disk.getSn().isEmpty()) desc.append("，缺少SN码");
            if (disk.getCapacity() == null) desc.append("，缺少容量");
            upsertViolation(null, null, disk.getId(), disk.getCurrentRecordId(), "incomplete_data", desc.toString());
        }

        List<DiskUsageRecord> incompleteRecords = usageRecordMapper.selectList(
                new LambdaQueryWrapper<DiskUsageRecord>()
                        .and(w -> w
                                .eq(DiskUsageRecord::getStatus, 1)
                                .and(w2 -> w2
                                        .isNull(DiskUsageRecord::getOutTime).or()
                                        .isNull(DiskUsageRecord::getOperatorId))
                                .or(i -> i.in(DiskUsageRecord::getStatus, 3, 4))
                                .and(w2 -> w2
                                        .isNull(DiskUsageRecord::getInTime).or()
                                        .isNull(DiskUsageRecord::getStorageContent).or()
                                        .eq(DiskUsageRecord::getStorageContent, "").or()
                                        .isNull(DiskUsageRecord::getOperatorId)))
        );
        for (DiskUsageRecord record : incompleteRecords) {
            StringBuilder desc = new StringBuilder("使用记录不完整");
            if (record.getOutTime() == null && record.getStatus() == 1) desc.append("，缺少出库时间");
            if (record.getInTime() == null && (record.getStatus() == 3 || record.getStatus() == 4)) desc.append("，缺少入库时间");
            if (record.getStorageContent() == null || record.getStorageContent().isEmpty()) desc.append("，缺少存储内容");
            if (record.getOperatorId() == null) desc.append("，缺少操作人");
            upsertViolation(null, null, record.getDiskId(), record.getId(), "incomplete_data", desc.toString());
        }
    }

    /**
     * 手动刷新违规记录：清空所有违规记录并重新扫描生成
     */
    @Transactional
    public int refreshViolations() {
        // 清空所有违规记录
        violationRecordMapper.delete(null);
        log.info("已清空所有违规记录");

        // 重新扫描各类违规
        int count = 0;

        // 1. 超时检查
        RuleConfig timeoutConfig = ruleConfigMapper.selectOne(
                new LambdaQueryWrapper<RuleConfig>()
                        .eq(RuleConfig::getRuleKey, "timeout_days"));
        if (timeoutConfig != null && timeoutConfig.getRuleValue() != null
                && !timeoutConfig.getRuleValue().trim().isEmpty()
                && timeoutConfig.getStatus() != null && timeoutConfig.getStatus() == 1) {
            try {
                int timeoutDays = Integer.parseInt(timeoutConfig.getRuleValue());
                LocalDateTime deadline = LocalDateTime.now().minusDays(timeoutDays);
                List<DiskUsageRecord> timeoutRecords = usageRecordMapper.selectList(
                        new LambdaQueryWrapper<DiskUsageRecord>()
                                .eq(DiskUsageRecord::getStatus, 1)
                                .lt(DiskUsageRecord::getOutTime, deadline));
                for (DiskUsageRecord record : timeoutRecords) {
                    String username = "";
                    SysUser user = sysUserMapper.selectById(record.getOperatorId());
                    if (user != null) username = user.getUsername();
                    recordViolation(record.getOperatorId(), username, record.getDiskId(), record.getId(),
                            "timeout", "硬盘出库超时，出库时间：" + record.getOutTime());
                    count++;
                }
            } catch (NumberFormatException e) {
                log.warn("timeout_days 配置值不是合法整数: {}", timeoutConfig.getRuleValue());
            }
        }

        // 2. 不完整数据检查
        if (getRuleBoolean("incomplete_data_check")) {
            List<HardDisk> incompleteDisks = hardDiskMapper.selectList(
                    new LambdaQueryWrapper<HardDisk>()
                            .and(w -> w
                                    .isNull(HardDisk::getModel).or().eq(HardDisk::getModel, "")
                                    .or().isNull(HardDisk::getSn).or().eq(HardDisk::getSn, "")
                                    .or().isNull(HardDisk::getCapacity))
            );
            for (HardDisk disk : incompleteDisks) {
                StringBuilder desc = new StringBuilder("硬盘信息不完整");
                if (disk.getModel() == null || disk.getModel().isEmpty()) desc.append("，缺少型号");
                if (disk.getSn() == null || disk.getSn().isEmpty()) desc.append("，缺少SN码");
                if (disk.getCapacity() == null) desc.append("，缺少容量");
                recordViolation(null, null, disk.getId(), disk.getCurrentRecordId(), "incomplete_data", desc.toString());
                count++;
            }

            List<DiskUsageRecord> incompleteRecords = usageRecordMapper.selectList(
                    new LambdaQueryWrapper<DiskUsageRecord>()
                            .and(w -> w
                                    .eq(DiskUsageRecord::getStatus, 1)
                                    .and(w2 -> w2
                                            .isNull(DiskUsageRecord::getOutTime).or()
                                            .isNull(DiskUsageRecord::getOperatorId))
                                    .or(i -> i.in(DiskUsageRecord::getStatus, 3, 4))
                                    .and(w2 -> w2
                                            .isNull(DiskUsageRecord::getInTime).or()
                                            .isNull(DiskUsageRecord::getStorageContent).or()
                                            .eq(DiskUsageRecord::getStorageContent, "").or()
                                            .isNull(DiskUsageRecord::getOperatorId)))
            );
            for (DiskUsageRecord record : incompleteRecords) {
                StringBuilder desc = new StringBuilder("使用记录不完整");
                if (record.getOutTime() == null && record.getStatus() == 1) desc.append("，缺少出库时间");
                if (record.getInTime() == null && (record.getStatus() == 3 || record.getStatus() == 4)) desc.append("，缺少入库时间");
                if (record.getStorageContent() == null || record.getStorageContent().isEmpty()) desc.append("，缺少存储内容");
                if (record.getOperatorId() == null) desc.append("，缺少操作人");
                recordViolation(null, null, record.getDiskId(), record.getId(), "incomplete_data", desc.toString());
                count++;
            }
        }

        log.info("违规记录刷新完成，共生成 {} 条", count);
        return count;
    }
